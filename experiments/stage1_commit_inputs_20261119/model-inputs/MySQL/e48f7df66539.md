## 提交身份与原始说明节选

数据库：MySQL

第一父版本：865e108e048e8eeb94027105395acd7e2d76d2b6

目标版本：e48f7df665392a46142efbb4928cb6b0ed1dbbf4

~~~text
When copying items into a temporary table row, we have different
handling based on whether the Item is an Item_field or not; we go and
look at the underlying Field, and do special handling of it. This causes
a bifurcated copy system, where everything needs to consider both
param->copy_fields and param->items_to_copy, and the identity of the
Item_field gets lost if one wishes to track Items through temporary
tables.

There's no good reason to do this, so just handle everything as
items_to_copy. The only thing copy_fields is used for now, is to copy
fields back and forth between window function's framebuffer tables;
it is needed there because it can also reverse the direction of copying.
~~~

bool copy_funcs(Temp_table_param *param, const THD *thd, Copy_func_type type) {
  DBUG_TRACE;
  if (!param->items_to_copy->size()) return false;

  Func_ptr_array *func_ptr = param->items_to_copy;
  uint end = func_ptr->size();
  for (uint i = 0; i < end; i++) {
    Func_ptr &func = func_ptr->at(i);
    Item *item = func.func();
    bool do_copy = false;
    switch (type) {
      case CFT_ALL:
        do_copy = true;
        break;
      case CFT_WF_FRAMING:
        do_copy = (item->m_is_window_function &&
                   down_cast<Item_sum *>(item)->framing());
        break;
      case CFT_WF_NON_FRAMING:
        do_copy = (item->m_is_window_function &&
                   !down_cast<Item_sum *>(item)->framing() &&
                   !down_cast<Item_sum *>(item)->needs_partition_cardinality());
        break;
      case CFT_WF_NEEDS_PARTITION_CARDINALITY:
        do_copy = (item->m_is_window_function &&
                   down_cast<Item_sum *>(item)->needs_partition_cardinality());
        break;
      case CFT_WF_USES_ONLY_ONE_ROW:
        do_copy = (item->m_is_window_function &&
                   down_cast<Item_sum *>(item)->uses_only_one_row());
        break;
      case CFT_HAS_NO_WF:
        do_copy = !item->m_is_window_function && !item->has_wf();
        break;
      case CFT_HAS_WF:
        do_copy = !item->m_is_window_function && item->has_wf();
        break;
      case CFT_WF:
        do_copy = item->m_is_window_function;
        break;
    }

    if (do_copy) {
      item->save_in_field(func.result_field(), /*no_conversions=*/true);
      /*
        Need to check the THD error state because Item::val_xxx() don't
        return error code, but can generate errors
        TODO: change it for a real status check when Item::val_xxx()
        are extended to return status code.
      */
      if (thd->is_error()) return true;
    }
  }
  return false;
}
