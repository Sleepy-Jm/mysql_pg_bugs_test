1331: /**
1332:  Replace columns in a condition that will be pushed to this derived table
1333:  with the derived table expressions.
1334: 
1335:  If there is a HAVING condition that needs to be pushed down, we replace
1336:  columns in the condition with references to the corresponding derived table
1337:  expressions and for WHERE condition, we replace columns with derived table
1338:  expressions.
1339: */
1340: 
1341: bool Condition_pushdown::replace_columns_in_cond(Item **cond, bool is_having) {
1342:   // For a view reference, the underlying expression could be shared if the
1343:   // expression is referenced elsewhere in the query. So we clone the expression
1344:   // before replacing it with derived table expression.
1345:   bool view_ref = false;
1346:   WalkItem((*cond), enum_walk::PREFIX, [&view_ref](Item *inner_item) {
1347:     if (inner_item->type() == Item::REF_ITEM &&
1348:         down_cast<Item_ref *>(inner_item)->ref_type() == Item_ref::VIEW_REF) {
1349:       view_ref = true;
1350:       return true;
1351:     }
1352:     return false;
1353:   });
1354:   Derived_table_info dti(m_derived_table, m_query_block);
1355: 
1356:   if (view_ref) {
1357:     (*cond) = (*cond)->transform(&Item::replace_view_refs_with_clone,
1358:                                  pointer_cast<uchar *>(&dti));
1359:     if (*cond == nullptr) return true;
1360:   }
1361:   Item *new_cond =
1362:       is_having ? (*cond)->transform(&Item::replace_with_derived_expr_ref,
1363:                                      pointer_cast<uchar *>(&dti))
1364:                 : (*cond)->transform(&Item::replace_with_derived_expr,
1365:                                      pointer_cast<uchar *>(&dti));
1366:   if (new_cond == nullptr) return true;
1367:   new_cond->update_used_tables();
1368:   (*cond) = new_cond;
1369:   return false;
1370: }
