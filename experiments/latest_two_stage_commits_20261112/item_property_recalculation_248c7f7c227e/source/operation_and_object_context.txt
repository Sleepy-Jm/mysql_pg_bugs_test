6501:   A transform creates a field representing the value of the derived table and
6502:   adds it as a hidden field to the select list.  Next, it replaces the subquery
6503:   in the item tree with this field.  If we replace in a HAVING condition, we
6504:   build an Item_ref, cf. PTI_simple_ident_ident::itemize which also creates a
6505:   Item_ref for a field reference in HAVING, because we may need to access the
6506:   field in a tmp table.

6508:   @param      thd       The session context
6509:   @param      subquery  The scalar subquery
6510:   @param      tr        The table reference for the derived table
6511:   @param      expr      The expression we are replacing (in)
6512: */

6514:                                            Table_ref *tr, Item **expr) {
6515:   if (!(*expr)->has_subquery()) return false;
6516: 
6517:   Item_singlerow_subselect::Scalar_subquery_replacement info(
6518:       subquery->item,
6519:       // make sure to not replace with one of the hidden fields, if present,
6520:       // e.g. for INTERSECT:
6521:       tr->table->field[tr->table->hidden_field_count], this,
6522:       subquery->m_add_coalesce);
6523: 
6524:   // ROLLUP wrappers might have been added to the expression at this point. Take
6525:   // care to transform the inner item and keep the rollup wrappers as is.
6526:   bool with_rollup_wrapper = is_rollup_group_wrapper(*expr);
6527:   Item *orig_unwrapped_item = unwrap_rollup_group(*expr);
6528:   Item *new_item = (*expr)->transform(&Item::replace_scalar_subquery,
6529:                                       pointer_cast<uchar *>(&info));

6533:   // retain its name so the metadata column name remains correct.
6534:   if (*expr != new_item) {
